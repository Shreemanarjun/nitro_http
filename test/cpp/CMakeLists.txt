# ─────────────────────────────────────────────────────────────────────────────
# nitro_http — C++ engine tests. Desktop only, opt in with
# `-DNITRO_HTTP_BUILD_TESTS=ON`.
#
# The engine is testable without a Dart VM because it only talks to Dart through
# two seams: `nitrohttp::installStreamSink` and `nitrohttp::setPostHook`. Both
# are replaced here with in-process fakes, so a test can assert on the exact
# bytes the bridge would have posted.
#
# `cpp-httplib` serves plain HTTP for the request/response, redirect, cookie and
# cache cases. It is fetched at a pinned tag rather than vendored: 10k lines of
# someone else's header in `git log` is a poor trade for a build-time download
# that CI caches anyway.
# ─────────────────────────────────────────────────────────────────────────────

cmake_minimum_required(VERSION 3.14)
project(nitro_http_tests LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
set(FETCHCONTENT_QUIET OFF)

FetchContent_Declare(
  googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz
  URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926
)
# Windows: keep gtest on the shared CRT so it matches the plugin's forced /MD.
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

FetchContent_Declare(
  httplib
  URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.3.tar.gz
  URL_HASH SHA256=a0567bcd6c3fe5cef1b329b96245119047f876b49e06cc129a36a7a8dffe173e
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
# Header-only: fetch the sources, do not run its CMake (which wants OpenSSL).
FetchContent_Populate(httplib)

set(NH_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/engine")
set(NH_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../lib/src/generated/cpp")
set(NH_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/native")
set(NH_DART_API_DL "${CMAKE_CURRENT_SOURCE_DIR}/../../src/dart_api_dl.c")

add_executable(nitro_http_engine_tests
  main_test.cpp
  support/TestSink.cpp
  support/HttpTestServer.cpp
  content_decoder_test.cpp
  wire_test.cpp
  arena_test.cpp
  cache_test.cpp
  ws_frame_test.cpp
  engine_test.cpp
  ${NITRO_HTTP_ENGINE_SOURCES}
  # `DartPost.cpp` references `Dart_PostCObject_DL` unconditionally. The symbol
  # lives in Dart's own DL shim, where it is a null function pointer until an
  # isolate initialises it — which is exactly the state the test binary wants:
  # `dartPostAvailable()` answers false and every post goes through the hook.
  "${NH_DART_API_DL}"
)

# `WsSession` drives a raw socket taken from curl with CONNECT_ONLY, so its
# fixture is a hand-written BSD-socket server rather than cpp-httplib. Only the
# POSIX spelling is implemented: these two files are the sole part of the suite
# that would need a winsock port, and the WebSocket path is not shipped on
# Windows yet.
if(NOT WIN32)
  target_sources(nitro_http_engine_tests PRIVATE
    support/WsTestServer.cpp
    ws_session_test.cpp
  )
endif()

target_include_directories(nitro_http_engine_tests PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}"
  "${NH_ENGINE_DIR}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src"
  "${NH_GENERATED_DIR}"
  "${NH_NATIVE_DIR}"
  "${httplib_SOURCE_DIR}"
)

# NITRO_HTTP_TESTING swaps `Dart_PostCObject_DL` for the hook, so the engine
# links without the Dart runtime present.
target_compile_definitions(nitro_http_engine_tests PRIVATE
  NITRO_HTTP_ENGINE_SEPARATE_TUS
  NITRO_HTTP_TESTING
  CPPHTTPLIB_THREAD_POOL_COUNT=4
)

target_link_libraries(nitro_http_engine_tests PRIVATE
  GTest::gtest
  nitro_http::curl
  nitro_http::platform
)

if(NOT MSVC)
  target_compile_options(nitro_http_engine_tests PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

# Opt-in AddressSanitizer. The zero-copy payload protocol is exactly the code a
# leak or double free hides in, so `-DNITRO_HTTP_TESTS_ASAN=ON` is the mode CI
# runs the arena and engine suites under.
option(NITRO_HTTP_TESTS_ASAN "Build the C++ engine tests with AddressSanitizer" OFF)
if(NITRO_HTTP_TESTS_ASAN AND NOT MSVC)
  target_compile_options(nitro_http_engine_tests PRIVATE
    -fsanitize=address -fno-omit-frame-pointer -g)
  target_link_options(nitro_http_engine_tests PRIVATE -fsanitize=address)
endif()

include(GoogleTest)
enable_testing()
# `enable_testing()` only marks THIS directory and below, and the tests live in
# a subdirectory of the `nitro_http` project. Without this the top-level build
# directory gets no CTestTestfile.cmake and `ctest --test-dir <build>` reports
# "No tests were found". src/CMakeLists.txt is regenerated by `nitrogen link`,
# so the flag is pushed up from here rather than added there.
set(CMAKE_TESTING_ENABLED TRUE PARENT_SCOPE)
# One CTest entry per gtest case: a failure names the exact case rather than
# "the test binary returned 1".
gtest_discover_tests(nitro_http_engine_tests
  DISCOVERY_TIMEOUT 60
  PROPERTIES TIMEOUT 120
)

# A second, tiny executable. `exit_path_test.cpp` explains why it cannot be a
# gtest case: the behaviour under test happens after `main` returns, so the exit
# status IS the assertion.
add_executable(nitro_http_exit_path_test
  exit_path_test.cpp
  ${NITRO_HTTP_ENGINE_SOURCES}
  "${NH_DART_API_DL}"
)
target_include_directories(nitro_http_exit_path_test PRIVATE
  "${NH_ENGINE_DIR}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src"
  "${NH_GENERATED_DIR}"
  "${NH_NATIVE_DIR}"
)
target_compile_definitions(nitro_http_exit_path_test PRIVATE
  NITRO_HTTP_ENGINE_SEPARATE_TUS
  NITRO_HTTP_TESTING
)
target_link_libraries(nitro_http_exit_path_test PRIVATE
  nitro_http::curl
  nitro_http::platform
)
if(NOT MSVC)
  target_compile_options(nitro_http_exit_path_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
if(NITRO_HTTP_TESTS_ASAN AND NOT MSVC)
  target_compile_options(nitro_http_exit_path_test PRIVATE
    -fsanitize=address -fno-omit-frame-pointer -g)
  target_link_options(nitro_http_exit_path_test PRIVATE -fsanitize=address)
endif()
add_test(NAME ExitPath.AnEngineStillAliveAtProcessExitDoesNotAbort
         COMMAND nitro_http_exit_path_test)
set_tests_properties(ExitPath.AnEngineStillAliveAtProcessExitDoesNotAbort
                     PROPERTIES TIMEOUT 60)
